@@@ filter cite-fields cite plain
function Cite(el)
  local parts = {}
  for i, c in ipairs(el.citations) do
    parts[#parts + 1] = pandoc.Str(i .. ":" .. c.id .. ":" .. c.mode .. ":" .. c.note_num .. ":" .. c.hash)
    parts[#parts + 1] = pandoc.Space()
    parts[#parts + 1] = pandoc.Emph(c.prefix)
    parts[#parts + 1] = pandoc.Strong(c.suffix)
  end
  return pandoc.Span(parts, {class = "cite"})
end

function Cite(el)

    c.mode = "SuppressAuthor"

    c.suffix = {pandoc.Str "end"}
    c.note_num = c.note_num + 5
    c.hash = 9
  end
  el.citations = el.citations
  return el
end

  for name in pairs(counts) do names[#names + 1] = name end
  table.sort(names)

  return pandoc.Pandoc({pandoc.CodeBlock(table.concat(lines, "\n"))}, doc.meta)
end

@@@ filter inlines-merge-strs plain rich
function Inlines(list)
  local out = pandoc.Inlines{}
  for _, el in ipairs(list) do
    local last = out[#out]
    if el.t == "Str" and last and last.t == "Str" then
      last.text = last.text .. el.text
    elseif el.t == "Space" and last and last.t == "Str" then
      last.text = last.text .. "_"
    else
      out:insert(el)
    end
  end
  return out
end

@@@ filter real-heading-shift rich nested
-- Moves each heading one level down, and turns level 6 into a paragraph.
function Header(h)
  if h.level >= 6 then
    return pandoc.Para(h.content)
  end
  h.level = h.level + 1
  return h
end
@@@ filter real-word-count rich nested
-- Adds the number of words of the body to the metadata field wordcount:
-- a text element with a character that is not punctuation is one word, and
-- inline code gives one word for each run without white space.
local total = 0
local function add_runs(text)
  for _ in text:gmatch('%S+') do total = total + 1 end
end
function Pandoc(doc)
  doc.blocks:walk {
    Inline = function(inline)
      if inline.t == 'Str' and inline.text:find('%P') then
        total = total + 1
      elseif inline.t == 'Code' then
        add_runs(inline.text)
      end
    end,
  }
  doc.meta.wordcount = tostring(total)
  return doc
end
@@@ filter real-link-rewrite rich nested
-- Rewrites links to pages of the site from .html to .md and marks external links.
function Link(link)
  if link.target:match('^https?://') then
    link.classes:insert('external')
  else
    link.target = link.target:gsub('%.html$', '.md')
  end
  return link
end
@@@ filter real-meta-reader meta rich
-- Puts the title and the authors of the metadata in front of the body.
function Pandoc(doc)
  local front = pandoc.Blocks{}
  if doc.meta.title then
    front:insert(pandoc.Header(1, doc.meta.title))
  end
  local authors = doc.meta.authors or doc.meta.author
  if authors then
    local names = pandoc.List{}
    for _, a in ipairs(authors) do
      local name = pandoc.utils.type(a) == 'table' and a.name or a
      names:insert(pandoc.utils.stringify(name))
    end
    front:insert(pandoc.Para(table.concat(names, ', ')))
  end
  doc.blocks = front .. doc.blocks
  return doc
end
@@@ filter real-table-builder meta plain
-- Builds a table of the metadata keys and their text.
function Pandoc(doc)
  local keys = {}
  for k in pairs(doc.meta) do keys[#keys + 1] = k end
  table.sort(keys)
  local rows = {}
  for _, k in ipairs(keys) do
    rows[#rows + 1] = pandoc.Row{
      pandoc.Cell(pandoc.Plain(k)),
      pandoc.Cell(pandoc.Plain(pandoc.utils.stringify(doc.meta[k]))),
    }
  end
  local head = pandoc.TableHead{pandoc.Row{pandoc.Cell(pandoc.Plain'Key'), pandoc.Cell(pandoc.Plain'Value')}}
  local body = {attr = pandoc.Attr(), body = rows, head = {}, row_head_columns = 0}
  local tbl = pandoc.Table(pandoc.Caption{pandoc.Plain'Metadata'},
    {{pandoc.AlignLeft, pandoc.ColWidthDefault}, {pandoc.AlignDefault, pandoc.ColWidthDefault}},
    head, {body}, pandoc.TableFoot())
  doc.blocks:insert(tbl)
  return doc
end
@@@ filter real-smallcaps-abbrev plain rich
-- Sets each word of two or more capital letters in small capitals.
function Str(el)
  local word, rest = el.text:match('^(%u%u+)(%p*)$')
  if word then
    return {pandoc.SmallCaps(word:lower()), pandoc.Str(rest)}
  end
end
@@@ filter real-remove-notes rich nested
-- Deletes footnotes and the space before them.
function Inlines(inlines)
  local result = pandoc.Inlines{}
  for i, el in ipairs(inlines) do
    if el.t == 'Note' then
      if result[#result] and result[#result].t == 'Space' then result:remove() end
    else
      result:insert(el)
    end
  end
  return result
end

end

@@@ filter real-quote-style cite rich
-- Replaces quoted text with the quotation marks of German typography.
function Quoted(q)
  local open, close = '\u{201E}', '\u{201C}'
  if q.quotetype == 'SingleQuote' then open, close = '\u{201A}', '\u{2018}' end
  local result = {pandoc.Str(open)}
  for _, el in ipairs(q.content) do result[#result + 1] = el end
  result[#result + 1] = pandoc.Str(close)
  return result
end

@@@ filter real-number-figures cite table
-- Numbers the figures and the tables and writes the number into each caption.
local figures, tables = 0, 0

  local first = caption.long[1]
  if first then
    first.content = {pandoc.Str(label), pandoc.Space()} .. first.content
  end
  return caption

end

@@@ filter real-cell-align table rich
-- Aligns the cells of the first column to the right, and counts the cells.
local cells = 0
function Table(tbl)
  for _, body in ipairs(tbl.bodies) do
    for _, row in ipairs(body.body) do
      for i, cell in ipairs(row.cells) do
        cells = cells + 1
        if i == 1 then cell.alignment = pandoc.AlignRight end
      end
    end
  end
  tbl.attributes.cells = tostring(cells)
  return tbl
end

@@@ filter tostring-each-kind rich cite table
-- Writes the text form of each element and each part of an element.
local seen = pandoc.List{}
-- An address in the text differs from run to run, so it is replaced.
local function note(x)
  if x ~= nil then seen:insert(pandoc.MetaString((tostring(x):gsub('0x%x+', 'ADDR')))) end
end
local function show(el)
  note(el)
  if el.attr then note(el.attr) end
  if el.t == 'Cite' then note(el.citations[1]) end
  if el.t == 'OrderedList' then note(el.listAttributes) end
  if el.t == 'Table' then
    note(el.caption); note(el.head); note(el.foot)
    note(el.head.rows[1]); note(el.head.rows[1] and el.head.rows[1].cells[1])
    note(el.colspecs[1] and pandoc.List(el.colspecs[1]))
  end
  if el.t == 'Figure' then note(el.caption) end
  if el.content and el.t ~= 'Table' then note(el.content) end
end
return {
  {Inline = show, Block = show},
  {Pandoc = function(d)
    note(d.meta)
    note(d)
    d.meta.shown = seen
    return d
  end},
}
